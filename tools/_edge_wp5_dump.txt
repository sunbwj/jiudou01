            method: 'POST', headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(body)

            set('wt-cols', tdoc.querySelectorAll('#wtTable thead th').length);
            set('wt-filters', ['#wtKeyword', '#wtRole', '#wtExec', '#wtStatus']
                .every(function (q) { return !!tdoc.querySelector(q); }) ? '1' : '0');
            set('wt-open-btns', tdoc.querySelectorAll('#wtTbody [data-open]').length);
            set('wt-exec-btns', tdoc.querySelectorAll('#wtTbody [data-exec]').length);
            set('wt-count', (tdoc.getElementById('wtCount') || {}).textContent || 'none');
            set('wt-empty', (tdoc.getElementById('wtTbody') || {}).textContent || 'none');
            set('wt-menu', (tdoc.body.textContent || '').indexOf('我的任务') > -1 ? '1' : '0');
            set('wt-no-projectplan', (tdoc.body.textContent || '').indexOf('项目计划') < 0 ? '1' : '0');
            /* 执行记录弹窗：真实打开并读取回填值 */
            if (tdoc.querySelector('#wtTbody [data-exec]')) {
                $t(tdoc.querySelector('#wtTbody [data-exec]')).trigger('click');
                await wait(700);
                var exDlg = tdoc.querySelector('.el-dialog');
                set('wt-exec-dlg', exDlg ? ['#wtExExec', '#wtExStatus', '#wtExActualStart',
                    '#wtExActualEnd', '#wtExRemark'].every(function (q) {
                        return !!exDlg.querySelector(q);
                    }) : 'no-dialog');
                set('wt-exec-tag-count', tdoc.querySelectorAll('#wtTbody .wp-exec-tag').length);
                var exOv = overlayWith(tdoc, '#wtExExec');
                if (exOv && exOv.querySelector('.btn-cancel')) {
                    $t(exOv.querySelector('.btn-cancel')).trigger('click');
                }
                await wait(300);
            } else {
                set('wt-exec-dlg', 'no-rows'); set('wt-exec-tag-count', 0);
            }

            /* ---- 列表页：11 列 / 启动按钮 / 我的任务入口 / 新建弹窗 ---- */
            await load(fr, '/workplan.html', 1800);
            var ldoc = fr.contentDocument, $l = fr.contentWindow.jQuery;
            set('list-cols', ldoc.querySelectorAll('.el-table thead th').length);
            set('list-start-btn', ldoc.querySelectorAll('#wpBtnStart').length);
            set('list-my-task-btn', ldoc.querySelectorAll('#wpBtnMyTask').length);
            set('list-status-badge', ldoc.querySelectorAll('.wp-status.is-on').length);
            $l('#wpBtnAdd').trigger('click');
            await wait(1000);
            var $proj = $l('#wpNewProject');
            var pov = $proj.find('option').map(function () { return String($l(this).val()); }).get();
            set('proj-option-count', pov.length);
            set('proj-empty-first', pov[0] === '');
            set('proj-has-real', pov.some(function (v) { return v.length === 36; }));
            set('proj-text', $proj.find('option').eq(1).text().slice(0, 40));
            var desc = ldoc.getElementById('wpNewDesc');
            set('desc-rect-h', desc ? Math.round(desc.getBoundingClientRect().height) : 'none');
            set('desc-rows', desc ? desc.rows : 'none');
            set('desc-hint-gone', ldoc.querySelectorAll('.wp-hintline').length === 0 &&
                (ldoc.body.textContent || '').indexOf('确定后进入 Visio 风格设计器') < 0);
            set('new-desc-tall', !!(desc && desc.getBoundingClientRect().height >= 120));
            set('js-errors', ERRS.length ? ERRS.join(' | ') : 'none');
            set('errors', 'none');
            set('done', '1');
        } catch (err) {
            set('errors', err && (err.message || err));
            set('js-errors', ERRS.length ? ERRS.join(' | ') : 'none');
            set('done', '0');
        } finally {
            cleanup();
        }
    }
    run();
})();
